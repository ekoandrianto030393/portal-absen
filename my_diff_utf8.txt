+function startKehadiranTicker() {
+    const container = document.getElementById('personnelRoster');
+    if (!container || container.dataset.scrollInitialized === 'true') return;
+    
+    if (container.scrollHeight <= container.clientHeight + 10) return;
+
+    container.dataset.scrollInitialized = 'true';
+    let isPaused = false;
+    container.onmouseenter = () => isPaused = true;
+    container.onmouseleave = () => isPaused = false;
+
+    let exactScroll = container.scrollTop;
+
+    function step() {
+        if (!isPaused && !window.isRosterUpdating) {
+            // [DIUBAH] Menghapus pergerakan lambat/halus. 
+            // Dibuat Jauh Lebih Cepat & Tegas agar lancar.
+            exactScroll += 3.0; 
+            container.scrollTop = exactScroll;
+            
+            // Cek Batas Bawah Pasti
+            if (Math.ceil(container.scrollTop + container.clientHeight) >= container.scrollHeight - 3) {
+                isPaused = true;
+                setTimeout(() => {
+                    exactScroll = 0;
+                    container.scrollTop = 0;
+                    isPaused = false;
+                }, 3000); // Jeda 3 detik di bawah
+            }
+        }
+        requestAnimationFrame(step);
+    }
